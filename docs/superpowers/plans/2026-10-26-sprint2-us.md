```ts
// backend/src/infrastructure/auth/jwt.ts
import { SignJWT, jwtVerify } from 'jose';
import { UnauthorizedError } from '../../domain/errors/domain-error.js';
import { isUserRole, type UserRole } from '../repositories/user-types.js';
import type { TokenClaims, TokenService } from './token-service.js';

const ISSUER = 'abp4-api';
const AUDIENCE = 'abp4-app';

export class JwtTokenService implements TokenService {
  private readonly secret: Uint8Array;

  constructor(secret: string, private readonly ttlSeconds: number) {
    this.secret = new TextEncoder().encode(secret);
  }

  async issue(claims: TokenClaims): Promise<{ token: string; expiresIn: number }> {
    const token = await new SignJWT({ role: claims.role })
      .setProtectedHeader({ alg: 'HS256' })
      .setSubject(claims.userId)
      .setIssuer(ISSUER)
      .setAudience(AUDIENCE)
      .setIssuedAt()
      .setExpirationTime(`${Math.max(this.ttlSeconds, 1)}s`)
      .sign(this.secret);
    return { token, expiresIn: Math.max(this.ttlSeconds, 1) };
  }

  async verify(token: string): Promise<TokenClaims> {
    try {
      const { payload } = await jwtVerify(token, this.secret, {
        issuer: ISSUER,
        audience: AUDIENCE,
        algorithms: ['HS256'],
      });
      if (typeof payload.sub !== 'string' || !isUserRole(payload.role)) {
        throw new UnauthorizedError('Token invalido');
      }
      return { userId: payload.sub, role: payload.role };
    } catch (err) {
      if (err instanceof UnauthorizedError) throw err;
      throw new UnauthorizedError('Token invalido');
    }
  }
}

export type { UserRole };
```

```ts
import { createHash, randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';

export function verifyPassword(password: string, stored: string): boolean {
  const parts = stored.split('$');
  if (parts.length !== 6 || parts[0] !== 'scrypt') return false;
  const [, n, r, p, salt, hash] = parts;
  if (!n || !r || !p || !salt || !hash) return false;
  const expected = Buffer.from(hash, 'hex');
  if (expected.length === 0) return false;
  const actual = scryptSync(password, salt, expected.length, {
    N: Number(n),
    r: Number(r),
    p: Number(p),
  });
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}
```

```ts
// backend/src/application/auth/login.use-case.ts
import { UnauthorizedError } from '../../domain/errors/domain-error.js';
import type { UserRepository } from '../../domain/ports/user-repository.js';
import { verifyPassword } from '../../shared/password-hasher.js';
import type { TokenService } from '../../infrastructure/auth/token-service.js';
import type { UserView } from './user-view.js';
import { toUserView } from './user-view.js';

const DUMMY_HASH = 'scrypt$16384$8$1$00000000000000000000000000000000$00';

export interface LoginOutput {
  accessToken: string;
  tokenType: 'Bearer';
  expiresIn: number;
  user: UserView;
}

export class LoginUseCase {
  constructor(private readonly users: UserRepository, private readonly tokens: TokenService) {}

  async execute(input: { email: string; password: string }): Promise<LoginOutput> {
    const user = await this.users.findByEmail(input.email.toLowerCase());
    // Verifica mesmo sem usuario para o tempo de resposta nao revelar se o e-mail existe.
    const passwordOk = verifyPassword(input.password, user?.passwordHash ?? DUMMY_HASH);
    if (!user || !passwordOk || !user.active) throw new UnauthorizedError('Credenciais invalidas');

    const { token, expiresIn } = await this.tokens.issue({ userId: user.id, role: user.role });
    return { accessToken: token, tokenType: 'Bearer', expiresIn, user: toUserView(user) };
  }
}
```

```ts
// backend/src/main/http/middlewares/auth.ts
import type { RequestHandler } from 'express';
import { ForbiddenError, UnauthorizedError } from '../../../domain/errors/domain-error.js';
import type { Container } from '../../composition-root.js';

export function requireAuth(container: Container): RequestHandler {
  return async (req, res, next) => {
    try {
      const [scheme, token] = (req.header('authorization') ?? '').split(' ');
      if (scheme?.toLowerCase() !== 'bearer' || !token) throw new UnauthorizedError('Token ausente');
      const claims = await container.tokens.verify(token);
      // Recarrega: desativacao e mudanca de perfil valem imediatamente.
      res.locals.user = await container.useCases.getCurrentUser.execute(claims.userId);
      next();
    } catch (err) {
      next(err);
    }
  };
}

export const requireAdmin: RequestHandler = (_req, res, next) => {
  if (res.locals.user?.role !== 'ADMIN') return next(new ForbiddenError('Perfil sem permissao'));
  next();
};
```

```ts
// backend/src/main/http/routes/auth.routes.ts
import { Router } from 'express';
import { LoginRequest, LoginResponse } from '../contract/schemas/auth.schemas.js';
import { User } from '../contract/schemas/users.schemas.js';
import type { Container } from '../../composition-root.js';
import { requireAuth } from '../middlewares/auth.js';

export function authRoutes(container: Container): Router {
  const router = Router();
  router.post('/auth/login', async (req, res) => {
    const body = LoginRequest.parse(req.body);
    res.status(200).json(LoginResponse.parse(await container.useCases.login.execute(body)));
  });
  router.get('/auth/me', requireAuth(container), (_req, res) => {
    res.status(200).json(User.parse(res.locals.user));
  });
```

```ts
// backend/src/main/mqtt/keep-alive.ts
import type { Logger } from '../../shared/logger.js';

export interface KeepAliveOptions {
  url: string;
  fetchImpl: typeof fetch;
  logger: Logger;
}

export interface Scheduler {
  every(intervalMs: number, task: () => void): () => void;
}

export const RENDER_KEEP_ALIVE_MS = 10 * 60_000;

export function createKeepAlive({ url, fetchImpl, logger }: KeepAliveOptions) {
  return {
    async ping(): Promise<void> {
      try {
        const response = await fetchImpl(`${url}/healthz`, { signal: AbortSignal.timeout(10_000) });
        if (!response.ok) logger.warn('keep-alive respondeu erro', { status: response.status });
      } catch (err) {
        // O plano free dorme por falta de trafego HTTP; falhar o ping nao pode derrubar o worker.
        logger.warn('keep-alive falhou', { error: err instanceof Error ? err.message : String(err) });
      }
    },
  };
}

export function scheduleKeepAlive(keepAlive: { ping: () => Promise<void> }, scheduler: Scheduler, intervalMs = RENDER_KEEP_ALIVE_MS): () => void {
  return scheduler.every(intervalMs, () => void keepAlive.ping());
}

export const systemScheduler: Scheduler = {
  every: (intervalMs, task) => {
    const timer = setInterval(task, intervalMs);
    timer.unref();
    return () => clearInterval(timer);
  },
};
```

```cpp
#pragma once
#include <stdint.h>

namespace qr_token {

struct State {
  char token[65] = {0};
  uint64_t expiresAt = 0;
  bool hasToken = false;
};

inline bool visible(const State& state, uint64_t nowEpoch, bool clockValid) {
  if (!state.hasToken || !clockValid) return false;
  if (state.token[0] == '\0') return false;
  return nowEpoch < state.expiresAt;
}

inline uint32_t secondsLeft(const State& state, uint64_t nowEpoch, bool clockValid) {
  if (!visible(state, nowEpoch, clockValid)) return 0;
  return static_cast<uint32_t>(state.expiresAt - nowEpoch);
}

}  // namespace qr_token
```

```cpp
  uint8_t version = 3;  // 29x29: cabe token de 64 caracteres com ECC LOW
  qrcode_initText(&qr, qrBuffer, version, ECC_LOW, token);
  const int scale = 6;
  const int side = qr.size * scale;
  const int left = (240 - side) / 2;
  const int top = 20;
  tft.fillRect(left - 8, top - 8, side + 16, side + 16, ILI9341_WHITE);
  for (uint8_t y = 0; y < qr.size; ++y) {
    for (uint8_t x = 0; x < qr.size; ++x) {
      if (qrcode_getModule(&qr, x, y)) {
        tft.fillRect(left + x * scale, top + y * scale, scale, scale, ILI9341_BLACK);
      }
    }
  }
}
}  // namespace
```

`showQr(state, now, clockValid)`: se visível e token mudou, redesenha o QR; desenha token e contagem (`expira em Ns`) embaixo; se não visível, apaga a área e escreve `SEM QR`. `update()` mantém o desenho do status (texto) embaixo, reaproveitando `display_text::render`.
