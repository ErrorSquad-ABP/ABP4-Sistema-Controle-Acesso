SELECT left(e.id::text,8) AS evento, e.method AS metodo,
 e.decision AS decisao, coalesce(e.reason,'-') AS motivo,
 coalesce(e."distanceMeters"::text,'-') AS distancia_m,
 coalesce(c."ackStatus",'-') AS ack
FROM "AccessEvent" e LEFT JOIN "TerminalCommand" c ON c."requestId"=e.id
WHERE e."occurredAt">'2026-11-11' ORDER BY e."occurredAt" DESC LIMIT 4;
